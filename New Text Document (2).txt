<script>
    let dataLog = JSON.parse(localStorage.getItem('hawa_log')) || [];
    let customItems = JSON.parse(localStorage.getItem('hawa_items')) || ["فروج مسحب كامل", "فخاد مسحب", "ورك مسحب", "صدر مسحب", "لحمة طاووق"];
    
    function initItems() {
        const select = document.getElementById('itemSelect');
        select.innerHTML = customItems.map(i => `<option value="${i}">${i}</option>`).join('');
    }

    function addNewItem() {
        const name = document.getElementById('newItemName').value;
        if (name) {
            customItems.unshift(name);
            localStorage.setItem('hawa_items', JSON.stringify(customItems));
            initItems();
            document.getElementById('newItemName').value = "";
        }
    }

    setInterval(() => { document.getElementById('clock').innerText = new Date().toLocaleTimeString('ar-LB'); }, 1000);
    initItems();
    renderTable();

    async function connectScale() {
        try {
            const port = await navigator.serial.requestPort();
            await port.open({ baudRate: 9600 });
            document.getElementById('scaleStatus').innerText = "متصل ✅";
            const textDecoder = new TextDecoderStream();
            port.readable.pipeTo(textDecoder.writable);
            const reader = textDecoder.readable.getReader();
            while (true) {
                const { value, done } = await reader.read();
                if (done) break;
                let val = value.replace(/[^\d.]/g, '');
                if (val) document.getElementById('totalWeight').value = val;
            }
        } catch (e) { alert("يرجى اختيار المنفذ"); }
    }

    function addEntry() {
        const weight = parseFloat(document.getElementById('totalWeight').value);
        const cages = parseInt(document.getElementById('cages').value) || 0;
        if (!weight) return;
        const entry = {
            time: new Date().toLocaleTimeString('ar-LB'),
            supervisor: document.getElementById('supervisor').value || "-",
            worker: document.getElementById('worker').value || "-",
            item: document.getElementById('itemSelect').value,
            total: weight,
            net: (weight - (cages * 1.5)).toFixed(2)
        };
        dataLog.push(entry);
        localStorage.setItem('hawa_log', JSON.stringify(dataLog));
        renderTable();
        document.getElementById('totalWeight').value = "";
    }

    function renderTable() {
        document.querySelector("#prodTable tbody").innerHTML = dataLog.map(r => `
            <tr><td>${r.time}</td><td>${r.supervisor}</td><td>${r.item}</td><td>${r.total}</td><td style="color:red; font-weight:bold;">${r.net}</td></tr>
        `).reverse().join('');
    }

    function downloadExcel() {
        const ws = XLSX.utils.json_to_sheet(dataLog);
        const wb = XLSX.utils.book_new();
        XLSX.utils.book_append_sheet(wb, ws, "الإنتاج");
        XLSX.writeFile(wb, `Report_Hawa_${new Date().toLocaleDateString()}.xlsx`);
    }

    function clearData() {
        if (prompt("أدخل كلمة المرور لتصفير السجل:") === "2021") {
            dataLog = [];
            localStorage.removeItem('hawa_log');
            renderTable();
        } else {
            alert("كلمة المرور خاطئة!");
        }
    }
</script>
